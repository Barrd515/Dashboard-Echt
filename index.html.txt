    <!-- Tailwind CSS -->

    <!-- React & ReactDOM -->
    <script src="https://unpkg.com/react@18/umd/react.production.min.js" crossorigin></script>
    <script src="https://unpkg.com/react-dom@18/umd/react-dom.production.min.js" crossorigin></script>
    
    <!-- Babel -->
    <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>
    
    <!-- FontAwesome -->
    <link rel="stylesheet" href="https://cdnjs.cloudflare.com/ajax/libs/font-awesome/6.4.0/css/all.min.css">

    <script>
        tailwind.config = {
            theme: {
                extend: {
                    colors: {
                        darkbg: '#0f172a',

                }
            }

    <style>
        .card-enter {
            animation: fadeIn 0.4s ease-out forwards;

        }
        @keyframes fadeIn {
            to { opacity: 1; transform: translateY(0); }
        }
        @keyframes spinSlow {

        .refreshing-icon {

</head>
<body class="bg-darkbg text-gray-100 font-sans antialiased min-h-screen">
    <div id="root"></div>

    <script type="text/babel">
        const API_KEY = 'd82f9k1r01qmgc0ftk2gd82f9k1r01qmgc0ftk30';
        
        // WICHTIG: Alle Symbole müssen US-Ticker sein, da Finnhub Free keine int. Börsen unterstützt.
        // Die USD-Preise werden im Code automatisch in EUR umgerechnet.
        const SYMBOLS = [
            'GOOGL',  // Alphabet
            'V',      // Visa
            'NET',    // Cloudflare
            'PANW',   // Palo Alto
            'ANET',   // Arista
            'SAP',    // SAP (US ADR)
            'ASML',   // ASML (US ADR)
            'MU',     // Micron
            'ETN',    // Eaton
            'MELI',   // MercadoLibre
            'NVDA',   // NVIDIA
            'BLK',    // Blackrock
            'JPM',    // JPMorgan
            'NEMTF',  // Nemetschek (US OTC Ticker)
            'NOW'     // ServiceNow
        ];

        const formatEur = (value) => {
            return new Intl.NumberFormat('de-DE', { style: 'currency', currency: 'EUR' }).format(value);
        };

        const formatPercent = (value) => {
            return new Intl.NumberFormat('de-DE', { style: 'percent', minimumFractionDigits: 2, maximumFractionDigits: 2 }).format(value / 100);
        };

        function StockCard({ stock, index }) {
            const { quote, profile, news, symbol, eurRate } = stock;
            
            // Sicherheitsabfragen (Null-Checks), falls die API leere Objekte zurückgibt
            const safeQuote = quote || {};
            const safeProfile = profile || {};
            const safeNews = news || [];

            const currentRaw = safeQuote.c || 0;
            const openRaw = safeQuote.o || 0;
            
            // Alle Aktien sind nun US-Ticker, daher rechnen wir alles von USD in EUR um
            const currentEUR = currentRaw * eurRate;
            const openEUR = openRaw * eurRate;
            
            const diffEUR = currentEUR - openEUR;
            const diffPercent = openRaw !== 0 ? ((currentRaw - openRaw) / openRaw) * 100 : 0;

            let trendClass = '';
            let shadowClass = '';
            let borderClass = '';
            let priceTextClass = '';
            let arrowIcon = '';
            let sign = '';

            if (diffEUR > 0) {
                trendClass = 'text-green-500';
                borderClass = 'border-green-500/40';
                shadowClass = 'shadow-[0_0_15px_rgba(34,197,94,0.3)]';
                priceTextClass = 'text-green-400';
                arrowIcon = 'fa-arrow-up';
                sign = '+';
            } else if (diffEUR < 0) {
                trendClass = 'text-red-500';
                borderClass = 'border-red-500/40';
                shadowClass = 'shadow-[0_0_15px_rgba(239,68,68,0.3)]';
                priceTextClass = 'text-red-400';
                arrowIcon = 'fa-arrow-down';
            } else {
                trendClass = 'text-yellow-500';
                borderClass = 'border-yellow-500/40';
                shadowClass = 'shadow-[0_0_15px_rgba(234,179,8,0.3)]';
                priceTextClass = 'text-yellow-400';
                arrowIcon = 'fa-arrows-up-down';
            }

            // Für die UI bereinigen wir das Symbol leicht (z.B. NEMTF -> NEM)
            const displaySymbol = symbol === 'NEMTF' ? 'NEM' : symbol;
            const companyName = safeProfile.name || (symbol === 'NEMTF' ? 'Nemetschek SE' : 'Unbekanntes Unternehmen');

            return (
                <div 
                    className={`card-enter bg-cardbg rounded-xl p-4 border border-gray-700 hover:border-opacity-0 hover:${borderClass} hover:${shadowClass} flex flex-col h-full transition-all duration-300 relative overflow-hidden`}
                    style={{ animationDelay: `${index * 0.05}s` }}
                >
                    <div className={`absolute inset-0 border-2 rounded-xl pointer-events-none opacity-30 ${borderClass} ${shadowClass}`}></div>

                    <div className="flex justify-between items-start mb-3 border-b border-gray-700/60 pb-3 z-10">
                        <div className="flex items-center space-x-3 min-w-0 pr-2">
                            <div className="w-10 h-10 bg-white rounded-full flex items-center justify-center p-1 flex-shrink-0 shadow-inner">
                                {safeProfile.logo ? (
                                    <img src={safeProfile.logo} alt={displaySymbol} className="max-w-full max-h-full object-contain" />
                                ) : (
                                    <span className="text-gray-800 font-bold text-sm">{displaySymbol[0]}</span>
                                )}
                            </div>
                            <div className="min-w-0">
                                <h2 className="text-lg font-bold tracking-wide text-white truncate leading-tight">{displaySymbol}</h2>
                                <p className="text-[11px] text-gray-400 truncate w-24" title={companyName}>{companyName}</p>
                            </div>
                        </div>
                        <div className="flex flex-col items-end flex-shrink-0">
                            <div className={`text-sm font-bold ${trendClass}`}>
                                {sign}{formatEur(diffEUR)}
                            </div>
                            <div className={`text-[11px] font-semibold ${trendClass} bg-gray-900/60 px-1.5 py-0.5 rounded mt-0.5`}>
                                {sign}{formatPercent(diffPercent)}
                            </div>
                        </div>
                    </div>

                    <div className="mb-4 z-10">
                        <div className="flex items-center justify-between">
                            <div className="flex items-center space-x-2">
                                <span className={`text-2xl font-extrabold tracking-tight ${priceTextClass}`}>
                                    {formatEur(currentEUR)}
                                </span>
                                <i className={`fa-solid ${arrowIcon} ${trendClass} text-lg`}></i>
                            </div>
                        </div>
                        <div className="flex items-center justify-between text-[11px] text-gray-500 mt-1">
                            <div>
                                <span className="uppercase tracking-wider mr-1">Eröffnung:</span>
                                <span className="text-gray-300 font-medium">{formatEur(openEUR)}</span>
                            </div>
                            <span className="text-[9px] bg-gray-800 px-1 rounded text-gray-500">USD Konvertiert</span>
                        </div>
                    </div>

                    <div className="mt-auto z-10">
                        <h3 className="text-[10px] font-bold text-gray-500 uppercase tracking-widest mb-2 flex items-center border-t border-gray-700/50 pt-3">
                            <i className="fa-regular fa-newspaper mr-1.5"></i> News
                        </h3>
                        <ul className="space-y-2">
                            {safeNews.length > 0 ? safeNews.map((item, idx) => (
                                <li key={idx} className="group">
                                    <a href={item.url} target="_blank" rel="noopener noreferrer" 
                                       className="flex text-xs text-gray-400 hover:text-blue-400 transition-colors duration-200">
                                        <span className="text-blue-500 mr-1.5 opacity-50 group-hover:opacity-100 flex-shrink-0 mt-0.5">▸</span>
                                        <span className="line-clamp-2 leading-snug">{item.headline}</span>
                                    </a>
                                </li>
                            )) : (
                                <li className="text-xs text-gray-600 italic">Derzeit keine aktuellen US-News verfügbar.</li>
                            )}
                        </ul>
                    </div>
                </div>
            );
        }

        function App() {
            const [stocksData, setStocksData] = React.useState([]);
            const [loading, setLoading] = React.useState(true);
            const [isRefreshing, setIsRefreshing] = React.useState(false);
            const [error, setError] = React.useState(null);
            const [progress, setProgress] = React.useState(0);
            const [lastUpdated, setLastUpdated] = React.useState(null);
            const [currentExchangeRate, setCurrentExchangeRate] = React.useState(0);

            const fetchData = async (isInitialLoad = true) => {
                if (isInitialLoad) setLoading(true);
                else setIsRefreshing(true);
                setError(null);

                try {
                    let liveEurRate = 0.92; // Fallback
                    try {
                        const fxRes = await fetch('https://open.er-api.com/v6/latest/USD').then(res => res.json());
                        if (fxRes && fxRes.rates && fxRes.rates.EUR) {
                            liveEurRate = fxRes.rates.EUR;
                        }
                    } catch (fxErr) {
                        console.warn("Konnte Live-Wechselkurs nicht laden, nutze Fallback.", fxErr);
                    }
                    
                    setCurrentExchangeRate(liveEurRate);

                    const today = new Date();
                    const past = new Date(today);
                    past.setDate(past.getDate() - 14); // Suchzeitraum für News etwas erweitern
                    
                    const toDate = today.toISOString().split('T')[0];
                    const fromDate = past.toISOString().split('T')[0];

                    const results = [];
                    
                    for (let i = 0; i < SYMBOLS.length; i++) {
                        const symbol = SYMBOLS[i];
                        
                        const quoteUrl = `https://finnhub.io/api/v1/quote?symbol=${symbol}&token=${API_KEY}`;
                        const profileUrl = `https://finnhub.io/api/v1/stock/profile2?symbol=${symbol}&token=${API_KEY}`;
                        const newsUrl = `https://finnhub.io/api/v1/company-news?symbol=${symbol}&from=${fromDate}&to=${toDate}&token=${API_KEY}`;

                        // Wir verwenden try-catch für jede API-Abfrage, damit ein Fehler bei einer Aktie nicht alles blockiert
                        let quoteRes = {}, profileRes = {}, newsRes = [];
                        try {
                            const [q, p, n] = await Promise.all([
                                fetch(quoteUrl).then(res => res.json()),
                                fetch(profileUrl).then(res => res.json()),
                                fetch(newsUrl).then(res => res.json())
                            ]);
                            quoteRes = q;
                            profileRes = p;
                            newsRes = n;
                        } catch(e) {
                            console.warn(`Fehler beim Abrufen der Daten für ${symbol}`);
                        }

                        results.push({
                            symbol,
                            quote: quoteRes,
                            profile: profileRes,
                            news: (Array.isArray(newsRes) ? newsRes : []).slice(0, 3), 
                            eurRate: liveEurRate
                        });

                        if (isInitialLoad) {
                            setProgress(Math.round(((i + 1) / SYMBOLS.length) * 100));
                        }
                        
                        // 200ms Delay um 60/Min API Limit strikt einzuhalten
                        await new Promise(r => setTimeout(r, 200));
                    }

                    setStocksData(results);
                    setLastUpdated(new Date().toLocaleTimeString('de-DE', { hour: '2-digit', minute:'2-digit', second:'2-digit' }));
                    setLoading(false);
                    setIsRefreshing(false);
                } catch (err) {
                    console.error(err);
                    if (isInitialLoad) {
                        setError(`Fehler beim Abrufen der Marktdaten. Das API Limit wurde eventuell überschritten. Bitte warte eine Minute.`);
                    }
                    setLoading(false);
                    setIsRefreshing(false);
                }
            };

            React.useEffect(() => {
                fetchData(true);
                const intervalId = setInterval(() => fetchData(false), 120000); // 2 Min
                return () => clearInterval(intervalId);
            }, []);

            if (loading) {
                return (
                    <div className="flex flex-col items-center justify-center min-h-screen">
                        <i className="fa-solid fa-satellite-dish fa-fade text-blue-500 text-6xl mb-6"></i>
                        <h2 className="text-xl font-semibold mb-2">Lade Daten für {SYMBOLS.length} Aktien...</h2>
                        <div className="w-72 bg-gray-800 rounded-full h-2 mt-4 overflow-hidden border border-gray-700">
                            <div className="bg-gradient-to-r from-blue-600 to-indigo-500 h-2 rounded-full transition-all duration-300" style={{ width: `${progress}%` }}></div>
                        </div>
                        <p className="mt-3 text-sm text-gray-500">{progress}% geladen (API-Schutz aktiv)</p>
                    </div>
                );
            }

            if (error) {
                return (
                    <div className="flex items-center justify-center min-h-screen px-4">
                        <div className="bg-red-500/10 border border-red-500/30 rounded-xl p-8 max-w-lg text-center">
                            <i className="fa-solid fa-ban text-red-500 text-5xl mb-4"></i>
                            <h2 className="text-lg font-bold text-red-400 mb-2">Verbindungsfehler oder Rate-Limit</h2>
                            <p className="text-sm text-gray-400 leading-relaxed">{error}</p>
                            <button onClick={() => window.location.reload()} className="mt-6 px-6 py-2.5 bg-gray-800 hover:bg-gray-700 text-white rounded-lg text-sm font-medium transition-colors border border-gray-700 shadow-lg">
                                Dashboard neu laden
                            </button>
                        </div>
                    </div>
                );
            }

            return (
                <div className="container mx-auto px-4 py-8 max-w-[1800px]">
                    <header className="mb-8 flex flex-col md:flex-row md:justify-between md:items-end border-b border-gray-800/80 pb-5">
                        <div>
                            <h1 className="text-3xl md:text-4xl font-extrabold tracking-tight mb-1">
                                Mein Portfolio <span className="text-transparent bg-clip-text bg-gradient-to-r from-blue-400 to-indigo-500">Dashboard</span>
                            </h1>
                            <p className="text-sm text-gray-400 flex items-center">
                                <i className="fa-solid fa-circle text-[8px] mr-2 text-green-500 animate-pulse"></i>
                                {SYMBOLS.length} Aktien im Live-Monitoring
                            </p>
                        </div>
                        <div className="mt-4 md:mt-0 flex flex-col items-start md:items-end">
                            <div className="text-xs text-gray-400 bg-gray-900/80 px-3 py-1.5 rounded-md inline-flex items-center border border-gray-800 shadow-sm">
                                {isRefreshing ? (
                                    <><i className="fa-solid fa-rotate-right refreshing-icon text-blue-400 mr-2"></i> Aktualisiere im Hintergrund...</>
                                ) : (
                                    <><i className="fa-regular fa-clock mr-1.5 text-gray-500"></i> Stand: <span className="text-gray-300 ml-1 font-semibold">{lastUpdated}</span></>
                                )}
                            </div>
                        </div>
                    </header>

                    <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 2xl:grid-cols-5 gap-4 lg:gap-5">
                        {stocksData.map((stock, index) => (
                            <StockCard key={stock.symbol} stock={stock} index={index} />
                        ))}
                    </div>
                    
                    <footer className="mt-16 text-center text-gray-600 text-sm pb-8 border-t border-gray-800 pt-8">
                        <p>Alle Preisangaben in Euro (€).</p>
                        <p className="mt-1 text-xs text-gray-500">US-Aktien werden in Echtzeit umgerechnet (Aktueller Wechselkurs: 1 USD = {currentExchangeRate.toFixed(4)} EUR).</p>
                    </footer>
                </div>
            );
        }

        const root = ReactDOM.createRoot(document.getElementById('root'));
        root.render(<App />);
    </script>
</body>
